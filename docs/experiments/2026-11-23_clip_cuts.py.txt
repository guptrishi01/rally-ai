import re, sys, statistics as st
t=[]; sc=[]
for line in open(sys.argv[1]):
    m = re.search(r"pts_time:([\d.]+)", line)
    if m: t.append(float(m.group(1))); continue
    m = re.search(r"scene_score=([\d.]+)", line)
    if m: sc.append(float(m.group(1)))
n=min(len(t),len(sc)); t=t[:n]; sc=sc[:n]
print("frames", n, "last t", t[-1])
s=sorted(sc); print("score percentiles 50/90/99/99.9/max:", *(round(s[int(len(s)*q)-1],4) for q in (.5,.9,.99,.999,1)))
# a cut = a score far above its local neighbourhood
for thr in (0.01,0.015,0.02,0.03,0.05,0.08):
    cuts=[t[i] for i in range(1,n-1) if sc[i]>thr and sc[i]>4*max(1e-4, st.median(sc[max(0,i-15):i]+sc[i+1:i+16]))]
    # collapse cuts within 0.5s
    c=[]
    for x in cuts:
        if not c or x-c[-1]>0.5: c.append(x)
    print(f"threshold {thr}: {len(c)} cuts -> {len(c)+1} clips")
