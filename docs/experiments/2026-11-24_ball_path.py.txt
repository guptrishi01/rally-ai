"""Prototype 2: streaming three-frame difference, ball-sized blobs solid, players faint.

Usage: trail2.py <outdir> <point_number> [...]
"""
import subprocess
import sys
import time
from pathlib import Path

import numpy as np

sys.path.insert(0, "src")
sys.path.insert(0, ".")
from swingvision_import import footage  # noqa: E402
from swingvision_import.config import ImportConfig  # noqa: E402
from swingvision_import.pipeline import SwingVisionImportPipeline  # noqa: E402
from swingvision_import.review import load_pending  # noqa: E402

OFFSET = 1.99
VIDEO = Path("full_match_video (1).mp4")
out = Path(sys.argv[1])
out.mkdir(parents=True, exist_ok=True)
work = out.parent / "fw"
cfg = ImportConfig(pending_dir=work / "pending", db_path=work / "x.db")
pipe = SwingVisionImportPipeline(cfg)
record = load_pending(next((work / "pending").glob("*.json")))
shots = pipe.shots_by_point(record)
exe = footage.ffmpeg_exe()
W, H, FPS = 1920, 1080, 30
BOX, SMALL = 15, 150


def crowded(mask, block=40, limit=220):
    h, w = mask.shape
    counts = mask.reshape(h // block, block, w // block, block).sum(axis=(1, 3))
    padded = np.pad(counts, 1)
    near = sum(
        padded[dy : dy + counts.shape[0], dx : dx + counts.shape[1]]
        for dy in range(3) for dx in range(3)
    )
    return np.repeat(np.repeat(near >= limit, block, axis=0), block, axis=1)


def trail(start, dur):
    proc = subprocess.Popen(
        [exe, "-hide_banner", "-loglevel", "error", "-ss", f"{start:.3f}", "-i", str(VIDEO),
         "-t", f"{dur:.3f}", "-an", "-vf", f"fps={FPS}", "-f", "rawvideo", "-pix_fmt", "rgb24", "-"],
        stdout=subprocess.PIPE,
    )
    size = W * H * 3
    total = int(dur * FPS)
    canvas = None
    greens = []
    index = 0
    while True:
        raw = proc.stdout.read(size)
        if len(raw) < size:
            break
        frame = np.frombuffer(raw, dtype=np.uint8).reshape(H, W, 3)
        if canvas is None:
            canvas = frame.astype(np.float32) * 0.6
        greens.append(frame[:, :, 1].astype(np.int16))
        if len(greens) == 3:
            a, b, c = greens
            moving = (np.abs(b - a) > 20) & (np.abs(b - c) > 20)
            if moving.any():
                small = moving & ~crowded(moving)
                big = moving & ~small
                hue = min(1.0, index / max(1, total - 1))
                colour = np.array(
                    [255 * hue, 255 * (1 - abs(hue - 0.5) * 2), 255 * (1 - hue)], dtype=np.float32
                )
                canvas[big] = canvas[big] * 0.7 + colour * 0.3
                # Thicken ball-sized blobs by one pixel so they survive downscaling.
                thick = small.copy()
                thick[1:, :] |= small[:-1, :]
                thick[:-1, :] |= small[1:, :]
                thick[:, 1:] |= small[:, :-1]
                thick[:, :-1] |= small[:, 1:]
                canvas[thick] = colour
            greens.pop(0)
        index += 1
    proc.wait()
    rgb = canvas.clip(0, 255).astype(np.uint8).tobytes()
    return subprocess.run(
        [exe, "-hide_banner", "-loglevel", "error", "-f", "rawvideo", "-pix_fmt", "rgb24",
         "-s", f"{W}x{H}", "-i", "-", "-frames:v", "1", "-q:v", "3", "-f", "image2", "-"],
        input=rgb, capture_output=True,
    ).stdout


for number in map(int, sys.argv[2:]):
    window = footage.shift_window(footage.plan_point(shots[number], pipe._host_name(record)), OFFSET)
    t = time.time()
    jpeg = trail(window.last_shot - 0.3, 3.3)
    (out / f"t2_{number}.jpg").write_bytes(jpeg)
    print(number, len(jpeg), f"{time.time() - t:.1f}s", flush=True)
